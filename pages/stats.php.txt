require_once(dirname(__DIR__) . DIRECTORY_SEPARATOR . 'config.php');

// Récupérer les statistiques
$stats = [
    'total_sorties' => 0,
    'total_prises' => 0,
    'poids_total' => 0,
    'prise_moyenne' => 0,
    'especes' => [],
    'meilleure_sortie' => null
];

try {
    // Total sorties
    $result = $pdo->query("SELECT COUNT(*) as total FROM sorties");
    $stats['total_sorties'] = $result->fetch(PDO::FETCH_ASSOC)['total'];
    
    // Total prises
    $result = $pdo->query("SELECT COUNT(*) as total FROM prises");
    $stats['total_prises'] = $result->fetch(PDO::FETCH_ASSOC)['total'];
    
    // Poids total
    $result = $pdo->query("SELECT SUM(poids) as total FROM prises WHERE poids IS NOT NULL");
    $row = $result->fetch(PDO::FETCH_ASSOC);
    $stats['poids_total'] = $row['total'] ? round($row['total'], 2) : 0;
    
    // Prise moyenne
    if ($stats['total_prises'] > 0) {
        $stats['prise_moyenne'] = round($stats['poids_total'] / $stats['total_prises'], 2);
    }
    
    // Espèces capturées
    $result = $pdo->query("
        SELECT espece, COUNT(*) as nombre, AVG(poids) as poids_moyen
        FROM prises 
        WHERE espece IS NOT NULL 
        GROUP BY espece 
        ORDER BY nombre DESC
    ");
    $stats['especes'] = $result->fetchAll(PDO::FETCH_ASSOC);
    
    // Meilleure sortie (plus de prises)
    $result = $pdo->query("
        SELECT s.id, s.date_sortie, s.lieu, COUNT(p.id) as nb_prises, SUM(p.poids) as poids_total
        FROM sorties s
        LEFT JOIN prises p ON s.id = p.sortie_id
        GROUP BY s.id
        ORDER BY nb_prises DESC
        LIMIT 1
    ");
    $stats['meilleure_sortie'] = $result->fetch(PDO::FETCH_ASSOC);
    
} catch (Exception $e) {
    $error = "Erreur lors du chargement des statistiques: " . $e->getMessage();
}
